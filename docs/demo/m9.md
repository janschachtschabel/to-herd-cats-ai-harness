```js
#!/usr/bin/env node
// A TCP relay between the API and the collab process that records, for every WebSocket
// upgrade the API sends upstream, which headers it carries (names, and for
// Authorization only the token's prefix). Values are never written. LISTEN=…, TARGET=…, OUT=…
import { appendFileSync } from "node:fs";
import net from "node:net";

const [listen, target, out] = [Number(process.env.LISTEN), Number(process.env.TARGET), process.env.OUT];
net
  .createServer((inbound) => {
    const upstream = net.connect(target, "127.0.0.1");
    let head = Buffer.alloc(0);
    let seen = false;
    inbound.on("data", (chunk) => {
      if (!seen) {
        head = Buffer.concat([head, chunk]);
        const end = head.indexOf("\r\n\r\n");
        if (end >= 0) {
          seen = true;
          const lines = head.subarray(0, end).toString("latin1").split("\r\n");
          const names = lines.slice(1).map((l) => l.split(":")[0].toLowerCase());
          const auth = lines.find((l) => /^authorization:/i.test(l)) ?? "";
          const scheme = auth.replace(/^authorization:\s*/i, "").slice(0, 15);
          appendFileSync(out, JSON.stringify({ request: lines[0].split(" ")[0] + " " + lines[0].split(" ")[1].split("?")[0], headers: names, authorization: scheme }) + "\n");
        }
      }
      upstream.write(chunk);
    });
    upstream.on("data", (chunk) => inbound.write(chunk));
    const end = () => { inbound.destroy(); upstream.destroy(); };
    inbound.on("close", end); upstream.on("close", end);
    inbound.on("error", end); upstream.on("error", end);
  })
  .listen(listen, "127.0.0.1");
```

```js
// --- an authenticator app (RFC 6238: HMAC-SHA1, 6 digits, 30 s), each step used once ---
const ALPHABET = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";
function base32(secret) {
  let bits = "";
  for (const char of secret.replace(/[\s=]/g, "").toUpperCase())
    bits += ALPHABET.indexOf(char).toString(2).padStart(5, "0");
  const bytes = [];
  for (let at = 0; at + 8 <= bits.length; at += 8) bytes.push(parseInt(bits.slice(at, at + 8), 2));
  return Buffer.from(bytes);
}
function hotp(secret, counter) {
  const message = Buffer.alloc(8);
  message.writeBigUInt64BE(BigInt(counter));
  const mac = createHmac("sha1", base32(secret)).update(message).digest();
  const offset = mac[mac.length - 1] & 0x0f;
  return String((mac.readUInt32BE(offset) & 0x7fffffff) % 10 ** 6).padStart(6, "0");
}

  }
}
```
