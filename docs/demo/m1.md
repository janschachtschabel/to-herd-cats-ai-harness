# Demo M1: Bereiche trennen Teams

Abnahme von Meilenstein 1 (Spezifikation 14): Zwei Personen in verschiedenen
Team-Bereichen sehen gegenseitig weder Inhalte noch Existenz, auch nicht per direkter
ID; Admins sehen beides, und ihr Zugriff steht im Audit.

## Ablauf

1. Die Admin-Person (die erste Inhaberin aus `HERD_BOOTSTRAP_*`) legt zwei Personen an,
   Anna und Ben (Systemrolle `member`), und zwei Team-Bereiche, „Team Nord“ und
   „Team Süd“.
2. Anna wird Mitglied von Team Nord, Ben von Team Süd. Die Admin-Person tritt aus
   beiden aus; sie sieht sie danach nur noch über ihr Admin-Recht, und genau dieser
   Zugriff wird auditiert (als Mitglied wäre er gewöhnlich).
3. Anna und Ben sehen in der Bereichsliste je nur ihren Team-Bereich. Den anderen
   öffnen sie auch per direkter ID nicht: Die API antwortet `404 space.not_found`
   (nicht 403, damit nicht einmal die Existenz verraten wird), die Mitgliederliste
   ebenso.
4. Die Admin-Person sieht beide Bereiche und öffnet beide. Das Audit enthält je
   Bereich einen Eintrag `space.admin_access`.

## Ausführen

Gegen eine laufende Compose-Installation (siehe README, Schnellstart); braucht `curl`
und `jq`. Das Skript legt echte Personen und Bereiche an, also nur auf einer
Test-Installation laufen lassen.

```bash
BASE=http://127.0.0.1:8080 ADMIN_EMAIL=admin@example.com ADMIN_PASSWORD=… bash demo-m1.sh
```

<details><summary><code>demo-m1.sh</code></summary>

```bash
#!/usr/bin/env bash
# M1 demo against a running compose installation. Needs curl and jq.
#   BASE=http://127.0.0.1:8080 ADMIN_EMAIL=admin@example.com ADMIN_PASSWORD=… bash demo-m1.sh
set -euo pipefail
: "${BASE:?}" "${ADMIN_EMAIL:?}" "${ADMIN_PASSWORD:?}"
jars="$(mktemp -d)"; trap 'rm -rf "$jars"' EXIT
H=(-H 'Content-Type: application/json' -H 'X-Requested-With: herd')

login() {  # name email password
  curl -sf -o /dev/null -c "$jars/$1" -X POST "$BASE/api/auth/login" "${H[@]}" \
    --data "$(jq -n --arg e "$2" --arg p "$3" '{email: $e, password: $p}')"
}
api() {  # who method path [body] -> prints "<status> <body>"
  local out; out="$(mktemp -p "$jars")"
  local code; code=$(curl -s -o "$out" -w '%{http_code}' -b "$jars/$1" -X "$2" "$BASE$3" "${H[@]}" \
    ${4:+--data "$4"})
  echo "$code $(cat "$out")"
}
body() { cut -d' ' -f2-; }
check() {  # description expected actual
  if [[ $2 == "$3" ]]; then echo "OK    $1"; else echo "FAIL  $1 (expected $2, got $3)"; exit 1; fi
}

echo "== 1. The admin creates two people and two team spaces"
login admin "$ADMIN_EMAIL" "$ADMIN_PASSWORD"
anna=$(api admin POST /api/admin/users \
  '{"email":"anna@example.com","name":"Anna Nord","password":"anna-password-1","system_role":"member"}' | body | jq -r .id)
ben=$(api admin POST /api/admin/users \
  '{"email":"ben@example.com","name":"Ben Süd","password":"ben-password-12","system_role":"member"}' | body | jq -r .id)
nord=$(api admin POST /api/spaces '{"name":"Team Nord","kind":"team"}' | body | jq -r .id)
sued=$(api admin POST /api/spaces '{"name":"Team Süd","kind":"team"}' | body | jq -r .id)
echo "Anna $anna, Ben $ben, Team Nord $nord, Team Süd $sued"

echo "== 2. Anna joins Team Nord, Ben joins Team Süd; the admin leaves both"
api admin PUT "/api/spaces/$nord/members" "{\"member_type\":\"user\",\"member_id\":\"$anna\",\"role\":\"admin\"}" >/dev/null
api admin PUT "/api/spaces/$sued/members" "{\"member_type\":\"user\",\"member_id\":\"$ben\",\"role\":\"admin\"}" >/dev/null
admin_id=$(api admin GET /api/auth/me | body | jq -r .user.id)
check "admin leaves Team Nord" 204 "$(api admin DELETE "/api/spaces/$nord/members/user/$admin_id" | cut -d' ' -f1)"
check "admin leaves Team Süd" 204 "$(api admin DELETE "/api/spaces/$sued/members/user/$admin_id" | cut -d' ' -f1)"

for who in anna:anna@example.com:anna-password-1:"Team Nord":"$nord":"$sued" \
           ben:ben@example.com:ben-password-12:"Team Süd":"$sued":"$nord"; do
  IFS=: read -r name email password own own_id other_id <<<"$who"
  echo "== 3. $name sees only $own"
  login "$name" "$email" "$password"
  teams=$(api "$name" GET '/api/spaces?limit=100' | body | jq -c '[.items[] | select(.kind=="team") | .name]')
  check "$name's team spaces" "[\"$own\"]" "$teams"
  check "$name opens her/his space by id" 200 "$(api "$name" GET "/api/spaces/$own_id" | cut -d' ' -f1)"
  other=$(api "$name" GET "/api/spaces/$other_id")
  check "$name opens the other space by id" "404 space.not_found" \
    "$(cut -d' ' -f1 <<<"$other") $(body <<<"$other" | jq -r .error.code)"
  check "$name lists the other space's members" 404 \
    "$(api "$name" GET "/api/spaces/$other_id/members" | cut -d' ' -f1)"
done

echo "== 4. The admin sees both, and the access is in the audit"
teams=$(api admin GET '/api/spaces?limit=100' | body | jq -c '[.items[] | select(.kind=="team") | .name] | sort')
check "admin's team spaces" '["Team Nord","Team Süd"]' "$teams"
check "admin opens Team Nord" 200 "$(api admin GET "/api/spaces/$nord" | cut -d' ' -f1)"
check "admin opens Team Süd" 200 "$(api admin GET "/api/spaces/$sued" | cut -d' ' -f1)"
api admin GET '/api/admin/audit?action=space.admin_access&limit=100' | body \
  | jq -r '.items[] | "\(.created_at)  \(.action)  \(.actor_name)  \(.target_id // "") \(.details // {} | tostring)"'
audited=$(api admin GET '/api/admin/audit?action=space.admin_access&limit=100' | body \
  | jq -r --arg n "$nord" --arg s "$sued" '[.items[] | select(.target_id==$n or .target_id==$s) | .target_id] | unique | length')
check "audit has admin access to both spaces" 2 "$audited"
echo "DEMO OK"
```
